-- 用户表
CREATE TABLE `users` (
  `id` int NOT NULL AUTO_INCREMENT,
  `username` varchar(64) UNIQUE,
  `password_hash` varchar(128),
  `nickname` varchar(64),
  `avatar` varchar(256),
  `gender` int DEFAULT 0 COMMENT '0-女，1-男',
  `age` int,
  `height` float,
  `weight` float,
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  INDEX `idx_username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 用户运动偏好表
CREATE TABLE `user_preferences` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` int,
  `preference` varchar(64) COMMENT '运动偏好类型',
  PRIMARY KEY (`id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 资讯分类表
CREATE TABLE `news_categories` (
  `id` varchar(32) NOT NULL,
  `name` varchar(64),
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 资讯表
CREATE TABLE `news` (
  `id` int NOT NULL AUTO_INCREMENT,
  `title` varchar(128),
  `content` text,
  `image` varchar(256),
  `source` varchar(64),
  `author` varchar(64),
  `views` int DEFAULT 0,
  `category_id` varchar(32),
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`category_id`) REFERENCES `news_categories` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 运动记录表
CREATE TABLE `sport_records` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` int,
  `type` varchar(32) COMMENT '运动类型',
  `distance` float COMMENT '运动距离（米）',
  `duration` int COMMENT '运动时长（秒）',
  `calories` float COMMENT '消耗卡路里',
  `start_time` datetime,
  `end_time` datetime,
  `avg_pace` varchar(16) COMMENT '平均配速',
  `route` text COMMENT '运动轨迹，JSON格式',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 运动目标表
CREATE TABLE `sport_goals` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` int,
  `daily_steps` int COMMENT '每日步数目标',
  `daily_distance` float COMMENT '每日距离目标（公里）',
  `daily_calories` float COMMENT '每日消耗卡路里目标',
  `weekly_frequency` int COMMENT '每周运动频次目标',
  `weekly_duration` int COMMENT '每周运动时长目标（分钟）',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  `updated_at` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 运动挑战表
CREATE TABLE `sport_challenges` (
  `id` int NOT NULL AUTO_INCREMENT,
  `title` varchar(128),
  `description` text,
  `start_date` date,
  `end_date` date,
  `goal_type` varchar(32) COMMENT '目标类型',
  `goal_value` float COMMENT '目标值',
  `reward_points` int COMMENT '奖励积分',
  `reward_badge` varchar(32) COMMENT '奖励徽章',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 运动挑战参与表
CREATE TABLE `challenge_participants` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` int,
  `challenge_id` int,
  `join_date` date,
  `progress` float DEFAULT 0 COMMENT '完成进度（百分比）',
  `is_completed` boolean DEFAULT false,
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`),
  FOREIGN KEY (`challenge_id`) REFERENCES `sport_challenges` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 场地表
CREATE TABLE `venues` (
  `id` int NOT NULL AUTO_INCREMENT,
  `name` varchar(128),
  `type` varchar(32),
  `location` varchar(256),
  `description` text,
  `image` varchar(256),
  `images` text COMMENT '多张图片，JSON格式',
  `capacity` int,
  `open_time` varchar(16),
  `close_time` varchar(16),
  `price` float,
  `facilities` text COMMENT '场地设施，JSON格式',
  `rules` text COMMENT '场地规则，JSON格式',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 场地时段表
CREATE TABLE `venue_timeslots` (
  `id` int NOT NULL AUTO_INCREMENT,
  `venue_id` int,
  `start_time` varchar(16),
  `end_time` varchar(16),
  `price` float,
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`venue_id`) REFERENCES `venues` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 场地预约表
CREATE TABLE `venue_reservations` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` int,
  `venue_id` int,
  `timeslot_id` int,
  `date` date,
  `participants` int,
  `remark` varchar(256),
  `status` varchar(16) DEFAULT 'pending' COMMENT 'pending, confirmed, cancelled, completed',
  `price` float,
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`),
  FOREIGN KEY (`venue_id`) REFERENCES `venues` (`id`),
  FOREIGN KEY (`timeslot_id`) REFERENCES `venue_timeslots` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 场地评价表
CREATE TABLE `venue_reviews` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` int,
  `venue_id` int,
  `rating` int COMMENT '1-5星',
  `comment` text,
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`),
  FOREIGN KEY (`venue_id`) REFERENCES `venues` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 社区帖子表
CREATE TABLE `community_posts` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` int,
  `type` varchar(32) COMMENT '帖子类型',
  `content` text,
  `images` text COMMENT '图片列表，JSON格式',
  `sport_record_id` int,
  `likes` int DEFAULT 0,
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`),
  FOREIGN KEY (`sport_record_id`) REFERENCES `sport_records` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 帖子评论表
CREATE TABLE `post_comments` (
  `id` int NOT NULL AUTO_INCREMENT,
  `post_id` int,
  `user_id` int,
  `content` text,
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`post_id`) REFERENCES `community_posts` (`id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 帖子点赞表
CREATE TABLE `post_likes` (
  `id` int NOT NULL AUTO_INCREMENT,
  `post_id` int,
  `user_id` int,
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`post_id`) REFERENCES `community_posts` (`id`),
  FOREIGN KEY (`user_id`) REFERENCES `users` (`id`),
  UNIQUE KEY `user_post_unique` (`user_id`, `post_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- AI教练聊天记录表
